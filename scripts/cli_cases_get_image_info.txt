# Command lines for the port of GetImageInfo (src/cli/get_image_info.rs),
# same format as cli_cases.txt.

# --- window title, source and alternate files, -z
R|-progress:%f -FileName t/images/Canon.jpg t/images/Nikon.jpg
R|-progress:%30b%p%%%r -FileName t/images/Canon.jpg t/images/Nikon.jpg
R|-srcfile %d%f.xmp -srcfile @ -FileName -Title t/images/XMP.jpg t/images/Canon.jpg

R|-z -FileName t/images/Canon.jpg
R|''
R|t/images/nonexistent.jpg t/images/Canon.jpg

# --- conditions, @JSON
R|-v -if '$Model +' -FileName t/images/Canon.jpg
R|-v -if '$Make eq "Canon"' -Model t/images/Canon.jpg t/images/Nikon.jpg
R|-fast -if '$Make' -Model t/images/Canon.jpg
R|-if '$Make' -if '$Model =~ /REBEL/' -Model t/images/Canon.jpg t/images/Nikon.jpg
R|-if 'warn "x\n"; 1' -Model t/images/Canon.jpg
R|-v2 -if 'warn "x\n"; 1' -Model t/images/Canon.jpg
R|-j '@JSON:{"Make":"A","Model":"B"}'
R|-s '@JSON:{"Make":"A","Model":"B"}'

# --- deleting and restoring originals
W|-Artist=x images/Canon.jpg -execute -v -restore_original images/Canon.jpg

W|-v -restore_original images/Nikon.jpg

# --- output files

W|-j -Model -w+ %dall.json images/Canon.jpg -execute -j -Model -w+ %dall.json images/Nikon.jpg
W|-X -Model -FileName -w %d%f.xml images/Canon.jpg images/Nikon.jpg
W|-p fmt_files/gpx.fmt -w %d%f.gpx images/GPS.jpg

W|-w %d%f.txt -XYZ images/Canon.jpg
W|-b -W %d%f_%t.%s -ThumbnailImage -PreviewImage images/Canon.jpg
W|-v2 -b -W %d%f_%t%c.%s -a -XMP:Subject images/XMP.jpg
W|-b -W+ %dall.dat -Model images/Canon.jpg images/Nikon.jpg
W|-v2 -b -W %d%f.%s -Wext jpg -ThumbnailImage -Model images/Canon.jpg
W|-v -Artist=x -w %d%f.txt images/Canon.jpg
